/* --- Flow lines --- */
// Clean, breathable density (no clutter!)
const NUM_LINES = window.innerWidth < 900 ? 40 : 80;

class FlowLine {
  constructor(idx){this.idx=idx;this.init();}
  init(){
    const spawn=Math.random();
    // 90% spawn beautifully just outside the edges to drift inward. 10% spawn inside.
    if(spawn<.3){this.x=Math.random()*W;this.y=-20;}
    else if(spawn<.55){this.x=W+20;this.y=Math.random()*H;}
    else if(spawn<.78){this.x=Math.random()*W;this.y=H+20;}
    else if(spawn<.9){this.x=-20;this.y=Math.random()*H;}
    else{this.x=Math.random()*W;this.y=Math.random()*H;}
    
    this.age=0;
    this.life=260+Math.random()*220;
    this.speed=.45+Math.random()*.9;
    this.angle=Math.random()*Math.PI*2;
    this.trail=[];
    this.maxTrail=100; // Slightly shorter tail so it doesn't get tangled and messy
    this.w=.3+Math.random()*.7;
    this.lightness=56+Math.floor(Math.random()*20);
    this.saturation=Math.random()>.4?58:10;
  }
  step(t){
    const nx=this.x/W, ny=this.y/H;
    const flow=(
      Math.sin(nx*3.2+t*.35)*Math.cos(ny*2.8-t*.28)+
      Math.sin(nx*1.4-t*.18)*Math.cos(ny*4+t*.22)*.5+
      Math.sin(nx*6+t*.12)*Math.cos(ny*5-t*.15)*.25
    )*Math.PI*.65;
    this.angle+=(flow-this.angle)*.042;

    if(mouse.active){
      const dx=mouse.x-this.x, dy=mouse.y-this.y;
      const d=Math.hypot(dx,dy);
      if(d<420){
        const pull=(1-d/420)*2.4;
        const ma=Math.atan2(dy,dx);
        this.angle+=Math.sin(ma-this.angle)*pull*.22;
      }
    }

    this.x+=Math.cos(this.angle)*this.speed;
    this.y+=Math.sin(this.angle)*this.speed;
    this.age++;
    this.trail.push({x:this.x,y:this.y});
    if(this.trail.length>this.maxTrail) this.trail.shift();
    
    // KILL BOUNDARY: Wait until they are 150px off-screen so they fade out gracefully without vanishing
    if(this.age>this.life||this.x<-150||this.x>W+150||this.y<-150||this.y>H+150) this.init();
  }
  draw(){
    if(this.trail.length<3)return;
    // The line naturally fades out as this.age reaches this.life
    const fade=Math.min(this.age/60,1)*Math.min((this.life-this.age)/60,1);
    ctx.beginPath();
    ctx.moveTo(this.trail[0].x,this.trail[0].y);
    for(let i=1;i<this.trail.length;i++){
      const a=this.trail[i-1],b=this.trail[i];
      ctx.quadraticCurveTo(a.x,a.y,(a.x+b.x)/2,(a.y+b.y)/2);
    }
    
    let h = 42;
    let s = this.saturation;
    let l = this.lightness; 
    let alpha = fade * 0.75; 

    if (isLightMode) {
      if (this.saturation === 10) {
        h = 25;
        s = 15;
        l = 15; 
        alpha = fade * 0.85; 
      } else {
        h = 34;
        s = 100;
        l = 35; 
        alpha = fade * 0.9; 
      }
    }

    ctx.strokeStyle=`hsla(${h},${s}%,${l}%,${alpha})`;
    ctx.lineWidth=this.w;
    ctx.lineCap='round';
    ctx.stroke();
  }
}

contactModal.addEventListener('click', (e) => {
  if (e.target === contactModal) {
    closeModal();
  }
});

contactForm.addEventListener('submit', function(e) {
  e.preventDefault(); 
  
  const originalText = submitBtn.textContent;
  submitBtn.textContent = 'Sending...';
  submitBtn.style.pointerEvents = 'none';

  fetch(contactForm.action, {
      method: "POST",
      body: new FormData(contactForm),
      headers: { 'Accept': 'application/json' }
  })
  .then(response => {
      if (response.ok) {
          const modalTitle = document.getElementById('modal-title');
          const successWrapper = document.getElementById('success-wrapper');

          modalTitle.style.transition = 'opacity 0.4s ease';
          contactForm.style.transition = 'opacity 0.4s ease';
          modalTitle.style.opacity = '0';
          contactForm.style.opacity = '0';

          setTimeout(() => {
              modalTitle.style.display = 'none';
              contactForm.style.display = 'none';
              
              successWrapper.style.display = 'block';
              setTimeout(() => {
                  successWrapper.style.opacity = '1';
              }, 50);
          }, 400);

          document.querySelectorAll('.input-group input, .input-group textarea').forEach(el => el.blur());
          contactForm.reset();
          
          setTimeout(() => {
              closeModal();
              
              setTimeout(() => {
                  successWrapper.style.opacity = '0';
                  successWrapper.style.display = 'none';
                  
                  modalTitle.style.display = 'block';
                  contactForm.style.display = 'block';
                  
                  setTimeout(() => {
                      modalTitle.style.opacity = '1';
                      contactForm.style.opacity = '1';
                  }, 50);
                  
                  submitBtn.textContent = originalText;
                  submitBtn.style.pointerEvents = 'all';
              }, 400); 
          }, 4500);

      } else {
          submitBtn.textContent = 'Error. Try Again.';
          submitBtn.style.pointerEvents = 'all';
          setTimeout(() => submitBtn.textContent = originalText, 3000);
      }
  })
  .catch(error => {
      submitBtn.textContent = 'Error. Try Again.';
      submitBtn.style.pointerEvents = 'all';
      setTimeout(() => submitBtn.textContent = originalText, 3000);
  });
});
